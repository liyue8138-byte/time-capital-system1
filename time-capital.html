            <!-- 左侧：时间投资表单 (8/12) -->

                    <h2 class="text-lg font-bold text-slate-700">时间分配与投资</h2>

                                <span class="w-3 h-3 rounded-full" :class="cat.color"></span>

                                   class="w-full h-2 bg-slate-200 rounded-lg appearance-none cursor-pointer accent-emerald-500">

                            <input type="number" min="0" max="24" step="0.5" 

                                   v-model.number="inputs[key]" @input="validateAndSave"

                                   class="w-16 border border-slate-200 rounded px-1.5 py-0.5 text-center text-sm font-semibold focus:outline-emerald-500">

                <div class="mt-6 flex justify-end">

                        清空今日投资

                    </button>

                </div>

            </div>


            <!-- 右侧：收益趋势分析 (5/12) -->

            <div class="lg:col-span-5 flex flex-col gap-6">

                <!-- 投资占比饼图 -->

                <div class="bg-white p-6 rounded-2xl shadow-sm border border-slate-100 flex flex-col items-center">

                    <h2 class="text-lg font-bold text-slate-700 self-start mb-4">今日投资配比</h2>

                        <canvas id="pieChart"></canvas>

    <script>

        const { createApp, ref, computed, onMounted, watch } = Vue;


        createApp({

            setup() {

                // 默认分类与颜色

                const categories = {

                    work: { name: '工作/事业', color: 'bg-indigo-500', hex: '#6366f1' },

                    study: { name: '学习/成长', color: 'bg-emerald-500', hex: '#10b981' },

                    health: { name: '健康/运动', color: 'bg-rose-500', hex: '#f43f5e' },

                    relation: { name: '人际关系', color: 'bg-sky-500', hex: '#0ea5e9' },

                    entertainment: { name: '娱乐/休息', color: 'bg-amber-500', hex: '#f59e0b' },

                    custom: { name: '自定义分类', color: 'bg-purple-500', hex: '#a855f7' }

                };


                // 初始化当前日期（格式：YYYY-MM-DD）

                const getTodayString = () => {

                    const d = new Date();

                    return d.toISOString().split('T')[0];

                };


                const selectedDate = ref(getTodayString());

                

                // 输入的数据绑定

                const inputs = ref({

                    work: 0,

                    study: 0,

                    health: 0,

                    relation: 0,

                    entertainment: 0,

                    custom: 0

                });


                // 核心计算属性

                const totalHours = computed(() => {

                    return Object.values(inputs.value).reduce((sum, val) => sum + (Number(val) || 0), 0);

                });


                const totalRevenue = computed(() => {

                    return Math.round(totalHours.value * 100);

                });


                const remainingHours = computed(() => {

                    return 24 - totalHours.value;

                });


                const remainingRevenue = computed(() => {

                    return Math.round(remainingHours.value * 100);

                });


                // 本地存储管理

                const loadDateData = () => {

                    const localData = localStorage.getItem('time_investment_data');

                    const allData = localData ? JSON.parse(localData) : {};

                    

                    if (allData[selectedDate.value]) {

                        inputs.value = { ...allData[selectedDate.value] };

                    } else {

                        // 默认新的一天全清空

                        inputs.value = { work: 0, study: 0, health: 0, relation: 0, entertainment: 0, custom: 0 };

                    }

                    updateCharts();

                };


                const validateAndSave = () => {

                    // 防止负数

                    for (const key in inputs.value) {

                        if (inputs.value[key] < 0) inputs.value[key] = 0;

                    }

                    

                    const localData = localStorage.getItem('time_investment_data');

                    const allData = localData ? JSON.parse(localData) : {};

                    allData[selectedDate.value] = { ...inputs.value };

                    localStorage.setItem('time_investment_data', JSON.stringify(allData));

                    

                    updateCharts();

                };


                const resetCurrentDay = () => {

                    inputs.value = { work: 0, study: 0, health: 0, relation: 0, entertainment: 0, custom: 0 };

                    validateAndSave();

                };


                const setToToday = () => {

                    selectedDate.value = getTodayString();

                    loadDateData();

                };


                // 图表实例引用

                let pieChartInstance = null;

                let trendChartInstance = null;


                // 初始化和更新图表

                const updateCharts = () => {

                    // 1. 更新饼图

                    const pieCtx = document.getElementById('pieChart').getContext('2d');

                    const pieData = Object.keys(categories).map(key => inputs.value[key] || 0);

                    const pieColors = Object.values(categories).map(cat => cat.hex);

                    const labels = Object.values(categories).map(cat => cat.name);


                    if (pieChartInstance) pieChartInstance.destroy();

                    

                    if (totalHours.value > 0) {

                        pieChartInstance = new Chart(pieCtx, {

                            type: 'doughnut',

                            data: {

                                labels: labels,

                                datasets: [{

                                    data: pieData,

                                    backgroundColor: pieColors,

                                    borderWidth: 0

                                }]

                            },

                            options: {

                                responsive: true,

                                plugins: { legend: { display: false } },

                                cutout: '70%'

                            }

                        });

                    }


                    // 2. 更新近7天趋势图

                    const trendCtx = document.getElementById('trendChart').getContext('2d');

                    const localData = localStorage.getItem('time_investment_data');

                    const allData = localData ? JSON.parse(localData) : {};


                    // 获取过去7天的日期

                    const labelsTrend = [];

                    const dataTrend = [];

                    for (let i = 6; i >= 0; i--) {

                        const d = new Date();

                        d.setDate(d.getDate() - i);

                        const dateStr = d.toISOString().split('T')[0];

                        // 转换标签为 MM-DD 格式便于移动端显示

                        labelsTrend.push(dateStr.slice(5)); 

                        

                        const dayData = allData[dateStr] || {};

                        const dayHours = Object.values(dayData).reduce((sum, val) => sum + (Number(val) || 0), 0);

                        dataTrend.push(dayHours * 100);

                    }


                    if (trendChartInstance) trendChartInstance.destroy();

                    trendChartInstance = new Chart(trendCtx, {

                        type: 'bar',

                        data: {

                            labels: labelsTrend,

                            datasets: [{

                                label: '日收益 (元)',

                                data: dataTrend,

                                backgroundColor: 'rgba(16, 185, 129, 0.8)',

                                borderRadius: 6

                            }]

                        },

                        options: {

                            responsive: true,

                            maintainAspectRatio: false,

                            plugins: { legend: { display: false } },

                            scales: {

                                y: { beginAtZero: true, max: 2400 }

                            }

                        }

                    });

                };


                // 生成演示用的历史数据（仅在本地完全无数据时触发，方便初次查看效果）

                const generateMockData = () => {

                    const localData = localStorage.getItem('time_investment_data');

                    if (!localData || localData === '{}') {

                        const mock = {};

                        for (let i = 10; i >= 0; i--) {

                            const d = new Date();

                            d.setDate(d.getDate() - i);

                            const dateStr = d.toISOString().split('T')[0];

                            mock[dateStr] = {

                                work: Math.round((Math.random() * 6 + 4) * 2) / 2,

                                study: Math.round((Math.random() * 2 + 1) * 2) / 2,

                                health: Math.round((Math.random() * 1 + 0.5) * 2) / 2,

                                relation: Math.round((Math.random() * 2) * 2) / 2,

                                entertainment: Math.round((Math.random() * 3 + 1) * 2) / 2,

                                custom: 0

                            };

                        }

                        localStorage.setItem('time_investment_data', JSON.stringify(mock));

                    }

                };


                onMounted(() => {

                    generateMockData();

                    loadDateData();

                });


                return {

                    categories,

                    selectedDate,

                    inputs,

                    totalHours,

                    totalRevenue,

                    remainingHours,

                    remainingRevenue,

                    loadDateData,

                    validateAndSave,

                    resetCurrentDay,

                    setToToday

                };

            }

        }).mount('#app');

    </script>
